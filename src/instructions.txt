# SSH to turtlebot
- Remote PC
    ifconfig -> get ip of PC (eg: 192.168.222.91)
    sudo nmap -sP 192.168.222.91/24 -> get ip of turtlebot (eg: 192.168.222.123)
    ssh turtle@192.168.222.123 -> Enter password: 1171

# Run turtlebot bringup
- Turtlebot
    export TURTLEBOT3_MODEL=burger
    ros2 launch turtlebot3_bringup robot.launch.py

# Run Paveway nodes
- Remote PC
    ros2 launch paveway_web web.launch.py
    ros2 launch paveway_vision vision.launch.py 
    ros2 run turtlebot3_teleop teleop_keyboard

# Start Dashboard
- Get correct ip
    hostname -I -> Use this to update endpoint in dashboard files
- Dashboard PC
    cd paveway_dashboard
    npm run dev
